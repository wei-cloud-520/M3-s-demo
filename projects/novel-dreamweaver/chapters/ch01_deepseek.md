```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    int width;
    int height;
    unsigned char *data;
    int layers;
} DreamFrame;

DreamFrame* init_frame(int w, int h) {
    DreamFrame *frame = (DreamFrame*)malloc(sizeof(DreamFrame));
    if (!frame) return NULL;

    frame->width = w;
    frame->height = h;
    frame->layers = 3;
    frame->data = (unsigned char*)malloc(w * h * frame->layers);

    if (!frame->data) {
        free(frame);
        return NULL;
    }

    memset(frame->data, 0, w * h * frame->layers);
    return frame;
}

void render_layer(DreamFrame *frame, int layer, unsigned char *pattern) {
    if (layer < 0 || layer >= frame->layers) return;

    int offset = layer * frame->width * frame->height;
    for (int y = 0; y < frame->height; y++) {
        for (int x = 0; x < frame->width; x++) {
            int idx = offset + y * frame->width + x;
            frame->data[idx] = pattern[y * frame->width + x];
        }
    }
}
```

编译通过了。运行就崩。

他把光标移到render_layer那行，手指在桌上敲了两下。循环嵌套没问题，边界检查做了，指针偏移量算对了吗？他看向offset那行——layer * width * height。如果frame是空的呢？如果init_frame失败了，这里就会直接写进null指针的地址空间。

他没有马上改。而是把render_layer里的三层循环结构重新看了一遍。那个嵌套让他想起什么——不是技术层面，是感官层面。像穿过一扇门，又穿过一扇门，再穿过一扇门。每层循环打开一个新的空间维度。

```c
typedef struct {

} Point3D;

typedef struct {

    float width;
    float height;

Bench* create_bench(float x, float y, float z) {
    Bench *bench = (Bench*)malloc(sizeof(Bench));
    if (!bench) return NULL;
    
    bench->position.x = x;
    bench->position.y = y;
    bench->position.z = z;
    bench->width = 2.0f;
    bench->height = 0.8f;
    bench->material = 0;  // 木质长椅
    
    return bench;
}

void place_object_on_bench(Bench *bench, int side) {
    // side: 0 = 左边扶手下方, 1 = 右边扶手下方
    Point3D target;
    target.x = bench->position.x + (side == 0 ? -0.8f : 0.8f);
    target.y = bench->position.y - 0.2f;
    target.z = bench->position.z;
    
    // 在目标位置生成占位符
    // 等待外部输入填充
}
```

最后一行的注释是他故意的。留一个占位符，等待外部输入。如果那个做梦的人真的把什么东西放在了长椅左边扶手下面，那个物体就会成为这段代码里的具体数据。
